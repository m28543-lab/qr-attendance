function checkIn() { if (!currentStudent) { alert("กรุณาค้นหานักเรียนก่อน"); return; } if (currentStudent.status === "มาเรียน") { alert("นักเรียนคนนี้เช็คชื่อแล้ว"); return; } fetch(WEB_APP_URL, { method: "POST", mode: "no-cors", headers: { "Content-Type": "text/plain;charset=utf-8" }, body: JSON.stringify({ studentId: currentStudent.id }) }) .then(function() { currentStudent.status = "มาเรียน"; showStudents(); alert( "เช็คชื่อสำเร็จ!\n\n" + currentStudent.name + "\nสถานะ: มาเรียน" ); document.getElementById("studentId").value = ""; document.getElementById("result").style.display = "none"; currentStudent = null; }) .catch(function(error) { console.log(error); alert("ไม่สามารถเชื่อมต่อกับ Google Sheets ได้"); }); }
